-- ============================================================ -- Task / Docs / Trip 페이지용 Supabase 테이블 + 스토리지 -- Supabase 대시보드 → SQL Editor 에서 전체 실행하세요. -- index.html / ledger.html과 같은 프로젝트, 같은 로그인 계정을 그대로 씁니다. -- ============================================================ -- ============ Task (포트폴리오 프로젝트) ============ create table if not exists tasks ( id text primary key, title text not null, description text, "notionUrl" text, "imageUrl" text, "updatedAt" text ); alter table tasks enable row level security; create policy "authenticated full access" on tasks for all using (auth.role() = 'authenticated') with check (auth.role() = 'authenticated'); -- task-images 버킷: 썸네일이라 공개로 두고(누구나 URL로 볼 수 있음), 업로드/삭제만 로그인 필요 insert into storage.buckets (id, name, public) values ('task-images', 'task-images', true) on conflict (id) do nothing; create policy "public read task-images" on storage.objects for select using (bucket_id = 'task-images'); create policy "authenticated upload task-images" on storage.objects for insert with check (bucket_id = 'task-images' and auth.role() = 'authenticated'); create policy "authenticated delete task-images" on storage.objects for delete using (bucket_id = 'task-images' and auth.role() = 'authenticated'); -- ============ Docs (중요 문서함) ============ create table if not exists documents ( id text primary key, title text not null, category text, memo text, "fileName" text, "storagePath" text, "uploadedAt" text ); alter table documents enable row level security; create policy "authenticated full access" on documents for all using (auth.role() = 'authenticated') with check (auth.role() = 'authenticated'); -- documents 버킷: 보험증서 등 민감한 파일이라 비공개, 로그인 사용자만 전체 접근(서명 URL로 다운로드) insert into storage.buckets (id, name, public) values ('documents', 'documents', false) on conflict (id) do nothing; create policy "authenticated access documents" on storage.objects for all using (bucket_id = 'documents' and auth.role() = 'authenticated') with check (bucket_id = 'documents' and auth.role() = 'authenticated');
